\section{Grundlagen}
\subsection{Zahlensysteme}
\begin{definition}
	Seien $X, Y$ zwei Mengen, dann ist die Menge aller geordneten Paare $(x,y)$ von Elementen $x \in X, y \in Y$ das kartesische Produkt $X \times Y$
\end{definition}
\begin{definition}
	Eine Operation oder Verknüpfung $\square$ in einer Menge $X$ ist eine Zuordnung, welche jedem Paar $(x,y)\in X\times X$ genaun ein Element $x\square y \in X$ zuordnet.
\end{definition}
\begin{definition}
	Eine Menge $X$ versehen mit einer Verknüpfung $\square$ heißt abelsche Gruppe, falls folgende Eigenschaften erfüllt sind:\begin{enumerate}
		\item Kommutativ
		\item Assoziativ
		\item Inverse Element ($\forall x \in X \exists \bar{x}\in X: x\square \bar{x}=e$)
		\item Neutrales Element ($\exists e \in X: \forall x\in X x\square e =x$)
	\end{enumerate}
\end{definition}
\begin{example}
	$(\{0,1\},+)$
\end{example}
\begin{theorem}
	Sei $(X,\square)$ eine abelsche Gruppe. Dann gilt:\begin{enumerateroman}
		\item Das neutrale Element ist eindeutig.
		\item Das inverse Element ist eindeutig.
		\item Das inverse Element des inversen Elements ist das Element selbst.
	\end{enumerateroman}
\end{theorem}
\prove
\begin{enumerateroman}
	\item Angenommen $\exists e, e' \in X: e\neq e'\implies e\square e'=e, e'\square
		e=e'\implies e'=e$ \lightning
	\item Angenommen $\exists \bar{x},\tilde{x} \in X: \bar{x}\neq\tilde{x}\implies
		\bar{x}=e\square \bar{x}=(\tilde{x}\square x) \square \bar{x}=
		\tilde{x}$\lightning
\end{enumerateroman}
\begin{definition}
	Eine Menge $X$ mit mindestens zwei Elementen und zwei Operationen $+$ und $\cdot$ heißt Körper, falls \begin{itemize}
		\item $(X,+)$ abelsche Gruppe
		\item $(X\setminus 0, \cdot)$ abelsche Gruppe
		\item Distributiv
	\end{itemize}
\end{definition}
\begin{definition}
	Ein Körper $(X,+,\cdot)$ mit einer Relation $<$ heißt archimedisch angeordneter Körper.\begin{enumerateroman}
		\item totale Ordnung: $\forall x,y \in X$ gilt genau eine der Möglichkeiten
		$x<y,x=y,x>y$
		\item Transitivität: $\forall x,y,z \in X: x<y,y<z\implies x<z$
		\item Monotonie bzgl. der Addition $\forall x,y,z \in X: x <y\implies x+z< y+z$
		\item Monotonie bzgl. der Multiplikation $\forall x,y,z \in X$ mit $z>0$:
		$x<y\implies x\cdot z<y\cdot z$
		\item archimedische Ordnung: Zu $a,b > 0 \exists n \in \N: b<\sum\limits_{i=1}^{n}a=n\cdot a$
	\end{enumerateroman}
\end{definition}
\begin{theorem}
	Sei $0$ das neutrale Element bzgl. $+$ und $1$ das neutrale Element bzgl. $\cdot$ in einem archimedischen angeordneten Körper, dann gilt $1>0$.
\end{theorem}
\prove
Angenommen es gilt $1>0$
\begin{align*}
	\implies & 1=0\text{\lightning}                         \\
	\implies & 1<0\implies 1+\bar{1}<0+\bar{1}              \\
	         & \implies 0<\bar{1}                           \\
	         & \implies 0\cdot \bar{1}< \bar{1}\cdot\bar{1} \\
	         & \implies 0 < 1\text{\lightning}
\end{align*}
Folglich $1>0$.\\
\paragraph{Bemerkung}
Aus den Körperaxiomen sowie den Ordnungseigenschaften lassn sich alle gängigen
Rechenregeln ableiten.
\begin{enumerate}
	\item $a\cdot 0 =0$
	\item $a\cdot b = 0 \iff a=0 \lor b=0$
	\item $a(-b)=(-a)b=-(ab)$
	\item $(-a)(-b)=ab$
	\item $a>0\implies -a<0\land \frac{1}{a}>0$
	\item $a>b>0\implies 0<\frac{1}{a}<\frac{1}{b}$
\end{enumerate}
\begin{theorem}[Bernoulli-Ungleichung]
	Sei $(X,+,\cdot)$ ein archimedisch angeordneter Körper, dann gilt für jedes $n\in\N$ und $x\geq -1$\begin{align*}
		(1+x)^n\geq 1+n\cdot x
	\end{align*}
\end{theorem}
\prove
\begin{itemize}
	\item[IA]$n=1, (1+x)^1\geq 1+x$
	\item[IB]$(1+x)^n\geq 1+nx$
	\item[IS]$(1+x)^{n+1}\geq(1+x)(1+nx)=1+(n+1)x+n\cdot x^2\geq 1+(n+1)x$
\end{itemize}
\begin{theorem}
	Sei $(X,+,\cdot)$ ein archimedisch angeordneter Körper. Dann gilt
	\begin{enumerateroman}
		\item $b>1, M>0, \exists n\in \N: b^n>M$
		\item $0<q<1, \forall\epsilon>0, \exists n\in N: q^n<\epsilon$
	\end{enumerateroman}
\end{theorem}
\prove
\begin{enumerateroman}
	\item $x=b-1>0$. Mit $b^n=(x+1)^n\geq 1+nx$. $\exists n\in \N:n\cdot x>M\implies b^n>M$
	\item $b=q^{-1}>1$, dann $\exists n\in \N$. $b^n>\frac{1}{\epsilon} \implies q^n>\epsilon$
\end{enumerateroman}
\begin{definition}[Ordnungsvollständigkeit]
	Seien $A,B\subset X$ zwei nichtleere Mengen, so dass $\forall a\in A, b\in B: a\leq b$. Dann $\exists c\in X: \forall a\in A, b \in B: a\leq c\leq b$
\end{definition}
\begin{definition}
	Die reellen Zahlen $\R$ sind ein archimedisch angeordneter Körper, welcher die Ordnungvollständigkeit erfüllt.
\end{definition}
\paragraph{Bemerkung} Problem:
\begin{enumerate}
	\item Gibt es die reellen Zahlen überhaupt?
	\item Sind die reellen Zahlen eindeutig
\end{enumerate}
\subsection{Supremum und Infimum}
\begin{definition}
	Sei $A\subset \R$ ein nichtleere Teilemenge, dann heißt $x \in \R$
	\begin{itemize}
		\item obere Schranke von $A$, falls $\forall a \in A: a\leq x$
		\item untere Schranke von $A$, falls $\forall a \in A: a\geq x$
		\item Maximum, falls obere Schranke und $x\in A$
		\item Minimum, falls untere Schranke und $x\in A$
		\item Supremum: kleinste obere Schranke
		\item Infimum: größte untere Schranke
	\end{itemize}
\end{definition}
\begin{definition}
	Eine Menge $A\subset\R$ heißt beschränkt, falls sie eine obere und eine untere Schranke hat.
\end{definition}
\begin{theorem}
	Jede nichtleere, beschränkte Menge $A\subset\R$ besitzt ein eindeutig bestimmtes Infimum $\inf A$ und Supremum $\sup A$.
\end{theorem}
\prove \
\begin{itemize}
	\item Existenz:\\ $B=\{x\in \R\mid \forall a \in A: x\geq a\}$. Nach Vorraussetzung
	      sind $A, B \in \R$ nicht leer und es gilt für alle $a \in A, b\in B: a \leq
		      b$\\
	      \begin{align*}
		      \xRightarrow[vollstaendigkeit]{Ordnungs} \exists c \in \R: \forall a\in A, b in B: a\leq c\leq b
	      \end{align*}
	\item Eindeutigkeit:\\ Angenommen $c\neq \tilde{c}$ seien zwei verschiedene Suprema
	      von $A$. Angenommen $c< \tilde{c}\implies \tilde{c}$ ist Supremum. \lightning
\end{itemize}
Schreibweise kein Supremum/ Infimum: $\pm \infty$
\begin{theorem}
	Eine Zahl $x\in \R$ ist genau dann Supremum der nach oben beschränkten Menge $A$, wenn sie die folgenden Eigenschaften besitzt:
	\begin{enumerateroman}
		\item $\forall a\in A: x \geq a$
		\item $\forall \epsilon > 0, \exists a \in A: a+\epsilon > x$
	\end{enumerateroman}
\end{theorem}
\prove \
\begin{itemize}
	\item[$\Rightarrow$]
	      \begin{enumerateroman}
		      \item nach Definition erfüllt
		      \item Angenommen gilt nicht, d.h. $\exists\epsilon>0\forall a\in A: a+\epsilon
			      \leq x\implies \forall a \in A: a \leq x-\epsilon$ \lightning
	      \end{enumerateroman}
	\item[$\Leftarrow$] Dann ist $x$ obere Schranke von $A$.\\ Angenommen $y<x$ wäre kleinere obere Schranke. Mit $\epsilon=x-y$ und $\forall a \in A$ gilt \begin{align*}
		      a+\epsilon=a+x-y
	      \end{align*}
	      Da $a<y\implies a-y<0\implies a+\epsilon\leq x$\lightning
\end{itemize}

\begin{theorem}[Intervallschachtellung]
	Es sei $I_n=[a_n, b_n], n\in \N$ ein System von Intervallen mit $I_{n+1}\subset I_n$. Dann ist $\bigcap_{n\in \N}I_n=\{x\in \R\mid x\in I_n \forall n\in \N\}$ nicht leer, d.h. $\exists c \in \R$ mit $a_n\leq c\leq b_n \forall n \in \N$. Existiert zu jedem $\epsilon >0$ ein $n\in\N$ mit $b_n-a_n<\epsilon$, dann gilt $\bigcap_{n\in\N} I_n = \{c\}$. $I_n$ heißt dann Intervallschachtellung zu der Zahl $c$.
\end{theorem}
\paragraph{Bemerkung} Es ist wichtig, dass die Intervalle abgeschlossen sind. \prove \begin{itemize}
	\item Existenz: Man setzt $A=\{a_n\mid n\in \N\}, B=\{b_n\mid n\in \N\}$. Dann gilt
	      $\forall k>n: a_n\leq a_k \leq b_k \leq b_n\implies a\leq b \forall a\in A,
		      b\in B$. Nach dem Vollständigkeitsaxiom existiert ein $c\in \R$ mit $a\leq
		      c\leq b \forall a\in A,b\in B\implies c \in[a_n,b_n] \forall n \in N \implies c
		      \in \bigcap_{n\in\N} I_n$
	\item Eindeutigkeit: Angenommen $c,\tilde{c}\in\bigcap_{n\in\N}I_n$ und $c\neq
		      \tilde{c}$. Setze für jedes $c< \tilde{c}, \epsilon =\tilde{c}-c>0$. Nach
	      Vorraussetzung $\exists n\in\N: b_n-a_n<\epsilon$\lightning zu $a_n\leq
		      c<\tilde{c}\leq b_n$
\end{itemize}
\begin{theorem}

\end{theorem}
\prove Sei dem Leser überlassen.

\section{Folgen und Reihen}
\begin{definition}[Zahlenfolge]
	Eine Zahlenfolge ist eine Abbildung von $\N \to \C$ $n\mapsto a_n$. Jedem $n \in \N$ wird ein $a_n \in \C$ zugeordnet.
\end{definition}
$(a_n)_{n\in \N}$ oder $a_1, a_2, \dots$
\begin{definition}
	Eine Folge $(a_n)$ heißt
	\begin{itemize}
		\item beschränkt $\iff \exists c\in \R: |a_n|\leq c\quad \forall n \in \N$
		\item oben beschränkt $a_n\leq c$
		\item unten beschränkt $a_n \geq c$
		\item $\sup a_n = \sup\{a_n\mid n\in \N\}$
		\item $\inf a_n = \inf\{a_n\mid n\in \N\}$
	\end{itemize}
\end{definition}
\begin{definition}
	Eine Zahl $a \in \R$ heißt Grenzwert einer Folge $(a_n)$, falls 
	\begin{align*}
		\forall \epsilon \in \C, \epsilon > 0, \exists n_0 \in \N: \forall n \geq n_0: |a_n-a|<\epsilon 
	\end{align*}
\end{definition}
Eine Folge heißt konvergent, falls sie einen Grenzwert hat. Ansonsten hei0t sie divergent.
\begin{align*}
	\lim_{n\to \infty} a_n = a
\end{align*}
Nullfolge
\begin{definition}
	Sei $\epsilon>0$. Die Menge $U_{\epsilon}=\{z\in \C\mid |z-a| < \epsilon\}$ heißt $\epsilon$-Umgebung von $a$.
\end{definition}
\paragraph{Bemerkung} $\epsilon$-Umgebung, fast alle enthalten
\begin{theorem}
	Der Grenzwert einer konvergenten Zahlenfolge ist eindeutig.
\end{theorem}
\prove Sei dem Leser überlassen.
\begin{theorem}
	Jede konvergente Folge ist beschränkt.
\end{theorem}
\prove Sei $a_n \to a$. Dann gilt für $\epsilon=1$, dass $a-1<a_n<a+1$ für alle bis auf endlich viele $a_n$. Seien $a_n, n\in N\subset \N$ die endlich viel $a_n$ für die das nicht gilt. Dann gilt für alle $n\in \N$
\begin{align*}
	a_n \leq \max\{a+1,a_n\mid n\in N\}
\end{align*}
Also ist $a_n$ beschränkt.
\subsection{Rechnen mit Grenzwerten}
\begin{lemma}
	Das Produkt einer beschränkten Folge $(a_n)$ mit einer Nullfolge $(b_n)$ ist eine Nullfolge.
\end{lemma}
\prove Sei $\epsilon >0$. Da $(a_n)$ beschränkt existiert $c>0$ mit $|a_n|<c$ für alle $n\in \N$. Da $(b_n)$ Nullfolge existiert $\forall \epsilon > 0\ \tilde{\epsilon}=\frac{\epsilon}{c}: |b_n|<\tilde{\epsilon}$.
\begin{align*}
	\implies \forall \epsilon > 0\ |a_nb_n|<c|b_n|<c\frac{\epsilon}{c}=\epsilon\implies \lim\limits_{n\to\infty}a_n b_n \to 0
\end{align*}
\begin{theorem}
	Seien $a_n \to a$ und $b_n \to b$ konvergente Folgen. Dann gilt:
	\begin{itemize}
		\item $a_n\pm b_n \to a\pm b$
		\item $a_n\cdot b_n\to a\cdot b$
		\item $\frac{a_n}{b_n}\to \frac{a}{b}$, falls $b_n, b \neq 0$
		\item $|a_n|\to |a|$
	\end{itemize}
\end{theorem}
\prove
\begin{itemize}
	\item Sei $\epsilon >0$. Da $a_n \to a$ und $b_n \to b$ existiert $n_a, n_b$ so, dass für alle $n\geq n_a$ gilt $|a-a_n|<\frac{\epsilon}{2}$ und $n\geq n_b$ gilt $|b-b_n|<\frac{\epsilon}{2}$. Für $n_0=\max\{n_a,n_b\}$ gilt dann für alle $n\geq n_0$
	\begin{align*}
		|a+b-a_n-bn|\leq |a-a_n|+|b-b_n| < \epsilon
	\end{align*}
	\item Es gilt
	\begin{align*}
		a_nb_n-ab&=(a_n-a)b_n+ab_n-ab\\
		&= (a_n-a)b_n +a(b_n-b)\\
		&\to 0 + 0\\
		a_nb_n &\to ab
	\end{align*}
	\item Es reicht zu zeigen, dass $\frac{1}{b_n}\to\frac{1}{b}$.
	\begin{align*}
		\frac{1}{b_n}-\frac{1}{b}&=(b-b_n)\cdot \frac{1}{b_nb}
	\end{align*}
	Reicht folglich zu zeigen, dass $\frac{1}{b_nb}$ bzw. $\frac{1}{b_n}$ beschränkt ist.
	Man setzt $\epsilon \frac{\epsilon}{2}$. Da $b_n to b$ $\exists n_0$ so, dass $|b_n-b|< \epsilon \forall n \geq n_0\implies |b_n| \geq |b|-|b-b_n|>|\frac{b}{2}| \implies \frac{1}{|b_n|}<\frac{2}{b}$.
	\item $||a_n|-|a||\leq |a_n-a|\to 0$
\end{itemize}
\begin{theorem}
	Seien $a_n\to a$, $b_n\to b$ zwei konvergente Folgen mit $a_n\geq b_n$, für fast alle $n\in \N$. Dann gilt $a\geq b$.
\end{theorem}
\prove Angenommen $b>a$. Sei $\epsilon=\frac{b-a}{2}>0$. Dann $\exists n_0\in \N: \forall n\geq n_0: |a_n-a|<\epsilon \land |b_n-b| <\epsilon$ Dann gilt für $N \geq n_0$:
\begin{align*}
	b_n-a_n&=(b_n-b)+(a-a_n)+b-a\\
	&\geq b-a-|b_n-b|-|a_n-a|>0 \iff b_n > a_n \text{\lightning}
\end{align*}
$\implies b\leq a$
\subsection{Monotonie}
\begin{definition}
	Eine reele $(a_n)$ heißt:
	\begin{enumerateroman}
		\item monoton wachsend, falls $\forall n\ a_{n+1}\geq a_n$
		\item monoton fallend, falls $\forall n\ a_{n+1}\leq a_n$
		\item monoton, falls monoton wachsend oder fallend
		\item streng monoton, falls $>$ oder $<$
	\end{enumerateroman}
\end{definition}
\begin{theorem}
	Jede monotone und beschränkte Zahlenfolge ist konvergent.
\end{theorem}
\prove Sei o.b.d.A $(a_n)$ monoton wachsend und beschränkt und $A$ die Menge der Folgenglieder von $(a_n)$. $A$ besitzt Supremum $a=\sup(A)$. Also $\forall \epsilon >0 \exists a_n\in A$ so, dass:
\begin{align*}
	a_n+\epsilon >a &\iff a-a_n < \epsilon\\
	a \geq a_n \to a_n-a \leq 0 < \epsilon \\
	\implies |a_n-a|< \epsilon
\end{align*}
Sei $n\geq n_0$. Da $(a_n)$ monoton wachsend folgt $a_{n_0}\leq a_n \leq a\implies |a-a_n|<\epsilon\implies a_n \to a$.
\begin{example}[Heron-Verfahren]
	Mit $a\geq 1$ definiert die nachfolgende Folge rekursiv:
	\begin{align*}
		x_1&=a\\
		x_{k+1}&=\frac{x_k+\frac{a}{x_k}}{2}
	\end{align*}
	Fragen:
	\begin{itemize}
		\item[(i)] Konvergenz?
		\item[(ii)] Wogegen?
	\end{itemize}
	\begin{itemize}
		\item[(ii)] Angenommen $(x_k)$ konvergiert gegen $x$:
		\begin{align*}
			x&=\frac{x+\frac{a}{x}}{2} \quad\text{(Fixpunkt)}\\
			x^2&=a\\
			x&=\sqrt{a}
		\end{align*}
		\item[[i)] Man zeige, dass $(x_k)$ monoton fallend und beschränkt.\\
		Es gilt $x_k>0 \forall k \in \N^+$ und:
		\begin{align*}
			x_{k+1}^2-a&=\frac{x_k^2+\frac{a^2}{x_k^2}+2a}{4}-a\\
			&=\frac{x_k^2+\frac{a^2}{x_k^2}-2a}{4}\\
			&=\left(\frac{x_k-\frac{a}{x_k}}{2}\right)^2\geq 0
			&\implies x_{k+1}^2\geq a \forall k \in \N^+
		\end{align*}
		Betrachte:
		\begin{align*}
			x_k-x_{k+1}&=x_k-\frac{x_k+\frac{a}{x_k}}{2}\\
			&=\frac{x_k^2-a}{2x_k} \geq 0\\
			&\implies x_k \geq x_{k+1}
		\end{align*}
		Also $(x_k)$ monoton fallend.
	\end{itemize}
\end{example}

\subsection{Binomialkoeffizienten}
\begin{definition}
	\begin{align*}
		z\in \C, k\in \N: \binom{n}{k}=\frac{\prod\limits_{z-k+1}^z}{k!}
	\end{align*}
\end{definition}
\paragraph{Pascalsches Dreieck} \ 
\begin{theorem}
	Für $z\in \C, k\in \N$:
	\begin{align*}
		\binom{z}{k}+\binom{z}{k+1}=\binom{z+1}{k+1}
	\end{align*}
\end{theorem}
\begin{theorem}[Binomischer Lehrsatz]
	Für $a,b\in \C, n\in \N$:
	\begin{align*}
		(a+b)^n=\sum\limits_{k=0}^na^kb^{n-k}
	\end{align*}
\end{theorem}
\begin{theorem}
	Für alle $k\in \N$ und $z\in \C$ mit $|z|>1$ gilt:
	\begin{align*}
		\lim\limits_{n\to \infty} \frac{n^k}{z^n} = 0
	\end{align*}
\end{theorem}
\prove Setze $x=|z|-1>0$. Dann gilt:\begin{align*}
	|z|^n=(1+x)^n=\sum\limits_{i=0}^n\binom{n}{i}x^i
\end{align*}
Für $n>2k+1$ gilt daher:
\begin{align*}
	|z|^n&\geq \binom{n}{k+1}x^{k+1}\\
	&\geq \left(\frac{n}{2}\right)^{k+1}\frac{x^{k+1}}{(k+1)!}\\
	\implies |\frac{n^k}{z^n}|&\leq \frac{n^k}{\left(\frac{n}{2}\right)^{k+1}\frac{x^{k+1}}{(k+1)!}}\\
	&=\frac{2^{k+1}(k+1)!}{nx^{k+1}}\to 0
\end{align*}
für $n\to \infty$.
